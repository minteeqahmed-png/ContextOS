React debounce hook

Delay search requests until typing pauses.

function useDebounce(value, delay = 300) {
  const [result, setResult] = useState(value);
  useEffect(() => {
    const timer = setTimeout(() => {
      setResult(value);
    }, delay);
    return () => clearTimeout(timer);
  }, [value, delay]);
  return result;
}
